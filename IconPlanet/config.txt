# IconPlanet - ตั้งค่า (บันทึกแล้วเปิดเกมใหม่)
# บรรทัดที่ขึ้นต้นด้วย # คือคอมเมนต์

# ความละเอียด texture ของแต่ละดาว (64-1024) ยิ่งสูงยิ่งคมแต่เปิดแมพครั้งแรกช้าลง
TextureSize=2048

# ไอคอนล็อกขนาดกับดาวจริงเสมอ แต่จะ "โผล่" เมื่อซูมเข้าจนดาวบนจอใหญ่ประมาณนี้ (พิกเซล เส้นผ่านศูนย์กลาง)
# ตอนซูมออกไกลกว่านี้เป็นจุด "●" ของเกมเดิม (ไม่คงไอคอนไว้) และจะสร้างไอคอนเฉพาะดาวที่ซูมเข้าใกล้เท่านั้น
MinVisiblePx=14

# ซูมเข้าจนรัศมีดาวบนจอเกินกี่พิกเซล ไอคอนจะค่อยๆ จางให้เห็นดาว/terrain จริงของเกม (กันภาพแตก)
# เริ่มจางที่ค่านี้ → จางหมดที่ 1.5 เท่า / 0 = ไม่จางเลย
# ค่าแนะนำ: 300–500 (จอ 1080p) / ยิ่งสูงยิ่งซูมเข้าได้ใกล้ก่อนไอคอนหาย
TerrainFadePx=1000

# true = จางดาว/ดวงจันทร์ตอนซูมออกไกลแบบเกมเดิม (ค่าเริ่มต้น) / false = ไม่จาง
FadeWhenZoomedOut=true

# แสดงเมฆด้านหน้าดาว (เฉพาะดาวที่มีข้อมูล FRONT_CLOUDS_DATA)
ShowClouds=true

# ลำดับการวาด (terrain บนแมพของเกมใช้ 10)
SortingOrder=11

# ขยายความสูงภูเขา/หลุมใน silhouette (1 = ตามจริง ตามจริงมักเล็กจนแทบมองไม่เห็นในไอคอน)
TerrainExaggeration=1

# โหมดไอคอน (กดปุ่มลัด P วนในเกม — เปลี่ยนปุ่มที่ Settings > Keybindings — และจำค่าล่าสุดให้เอง)
#   smooth  = น้ำทึบสีของแมพทับ terrain + terrain สี mapColor เห็นรายละเอียดนิดหน่อย + เมฆจางๆ (ค่าเริ่มต้น)
#   high    = รายละเอียดทั้งหมด terrain / น้ำ / เมฆด้านหน้า / Fog
#   default = วงกลมเดิมของเกม
Mode=smooth

# ชนิด mask มหาสมุทรของไฟล์ดาว: auto = ตรวจเอง (เทียบกับความสูง terrain จริง) / dark = ดำคือทะเล / light = ขาวคือทะเล / texture = ใช้สีฟ้าของ texture ดาว (ถ้าน้ำยังผิด ลองสลับค่าพวกนี้ทีละอัน)
WaterMask=auto

# true = ยกน้ำจริงของเกมให้วาดเหนือ terrain ในแมพ (ปกติน้ำอยู่ใต้ terrain เลยมองไม่เห็น) ทำให้ซูมเข้าจนไอคอนจางแล้วยังเห็นน้ำบน terrain จริง
RaiseGameWater=true

# ทดลอง: true = ซ่อนเมช terrain/น้ำของเกมใต้ดาวขณะที่ไอคอนแสดง (ไว้เช็กว่าเมชของเกมวาดทับไอคอนอยู่ไหม)
HideGameMesh=false

# ตรวจสอบ: none / mask (mask มหาสมุทรดิบ) / texture (texture ดาวดิบ) / clouds (เมฆอย่างเดียว) — ใช้ถ่ายภาพส่งให้ผู้ช่วยดู
DebugView=none

ShowWater=true
ShowRings=true

# true = ใช้ texture ดาวเป็นรูปจานเต็มดวง / false = ใช้เฉพาะสีตามมุมรอบขอบดาว
# ถ้าไอคอนบางดาวดูแปลกๆ (เป็นแฉกๆ หรือสีเพี้ยน) ลองสลับค่านี้
FullDiscTexture=true
